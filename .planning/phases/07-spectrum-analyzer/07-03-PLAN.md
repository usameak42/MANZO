Add this method after applyStyle():
```swift
    // MARK: - Phase 7: Spectrum View Integration (D-01, D-02)

    /// Adds ManzoSpectrumView as a subview of bodyView and applies the NeoAeroContainer
    /// chrome mask cutout. Call after applyStyle() so the NeoAeroContainer layer is present.
    ///
    /// Layout (D-01):
    ///   - Width: 225 pt (75 bars × 3 pt/bar — exact Winamp singlesize at 2× scale)
    ///   - Height: 32 pt (16 height levels × 2 pt/level)
    ///   - Position: centered horizontally in bodyView (275 pt wide), bottom edge flush to bodyView bottom
    ///
    /// Chrome mask (D-02):
    ///   The NeoAeroContainer layer gets a CAShapeLayer mask whose path is:
    ///   full bodyView rect MINUS the spectrumView frame (evenOdd fill rule = punch-through).
    func addSpectrumView(_ view: ManzoSpectrumView) {
        // Remove any previously added spectrum view (idempotent re-call safety).
        spectrumView?.removeFromSuperview()
        spectrumView = view

        view.translatesAutoresizingMaskIntoConstraints = false
        bodyView.addSubview(view)

        // D-01: center-horizontal in bodyView (275 pt wide), bottom-anchored, fixed 225×32 pt.
        NSLayoutConstraint.activate([
            view.widthAnchor.constraint(equalToConstant: 225),
            view.heightAnchor.constraint(equalToConstant: 32),
            view.centerXAnchor.constraint(equalTo: bodyView.centerXAnchor),
            view.bottomAnchor.constraint(equalTo: bodyView.bottomAnchor),
        ])

        // Force layout now so spectrumView.frame is computed before we read it for the mask.
        bodyView.layoutSubtreeIfNeeded()

        // D-02: Apply CAShapeLayer mask to NeoAeroContainer layer.
        // The mask path = full bodyView bounds rect MINUS the spectrumView frame in bodyView coords.
        // evenOdd fill rule: two overlapping rects cancel → transparent hole for spectrum window.
        applySpectrumChromeMask(spectrumFrame: view.frame)

        NSLog("MANZO Phase 7: ManzoSpectrumView added — frame=%@, bodyView=%@",
              NSStringFromRect(view.frame), NSStringFromRect(bodyView.bounds))
    }

    /// Punches a rectangular hole in the NeoAeroContainer chrome for the spectrum view.
    /// Called once after layout; call again if bodyView bounds change (rare).
    private func applySpectrumChromeMask(spectrumFrame: CGRect) {
        // Find the NeoAeroContainer layer in bodyView's sublayer tree.
        // Phase 6 inserts it with name "NeoAeroContainer" (ManzoRootView.applyStyle / D-09).
        guard let neoAeroLayer = bodyView.layer?.sublayers?.first(where: {
            ($0.value(forKey: "name") as? String) == "NeoAeroContainer"
        }) else {
            NSLog("MANZO Phase 7: applySpectrumChromeMask — NeoAeroContainer not found in bodyView; skipping mask")
            return
        }

        // Build mask path: full bodyView rect with spectrum rect cut out (even-odd rule).
        let maskPath = CGMutablePath()
        maskPath.addRect(bodyView.bounds)        // outer boundary (filled)
        maskPath.addRect(spectrumFrame)          // inner hole (cancelled by evenOdd rule)

        let maskLayer = CAShapeLayer()
        maskLayer.path = maskPath
        maskLayer.fillRule = .evenOdd
        maskLayer.frame = bodyView.bounds

        // Set NeoAeroContainer.mask — all rendering outside the path is clipped.
        // Note: CALayer.mask clips the layer itself (shows layer only where mask is opaque).
        // evenOdd fill in maskPath means: opaque everywhere EXCEPT the spectrum rect hole.
        neoAeroLayer.mask = maskLayer

        NSLog("MANZO Phase 7: NeoAeroContainer chrome mask applied — spectrumFrame=%@",
              NSStringFromRect(spectrumFrame))
    }
```
