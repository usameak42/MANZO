import AppKit
import Metal
import MetalKit

// ManzoSpectrumView: MTKView subclass for the real-time spectrum analyzer.
//
// Dimensions: 225 × 32 pt (75 bars × 3 pt/bar; 32 pt = 16 height levels × 2 pt/level).
// Pixel format: .rgba16Float via CAMetalLayer (NOT MTKView.colorPixelFormat — SPEC-01).
// Colorspace: displayP3 set explicitly — MTKView does NOT auto-configure P3 (spike 010 landmine).
// Render loop: CADisplayLink on a dedicated background thread (SPEC-04).
//
// Architecture note: ManzoSpectrumView holds a weak reference to the Rust handle pointer
// passed from AppDelegate. The handle's lifetime is owned by AppDelegate.
class ManzoSpectrumView: MTKView {

    // MARK: - Metal objects

    private var commandQueue: MTLCommandQueue!
    private var barPipelineState: MTLRenderPipelineState!
    private var peakPipelineState: MTLRenderPipelineState!

    // MARK: - Buffers

    // fftBuffer: shared storage for 75 float32 bar heights (range [0.0, 1.0]).
    // Filled each frame by calling manzo_get_spectrum via FFI.
    // storageModeShared: CPU-writable and GPU-readable without explicit copy (SPEC-02).
    private var fftBuffer: MTLBuffer!

    // barVertexBuffer: 75 bars × 4 vertices × 8 bytes (2× float2 per vertex: position + barIndex/height).
    // Extended by 3σ=6 pt on all sides for SDF bloom (CONTEXT.md D-03 / spike 011).
    private var barVertexBuffer: MTLBuffer!

    // peakVertexBuffer: 75 peak dots × 4 vertices (1-pt dot quads, extended for bloom).
    private var peakVertexBuffer: MTLBuffer!

    // MARK: - Peak-hold state (D-14, D-15, D-16)

    // Peak values and velocities in normalized [0.0, 1.0] space (matching FFT output).
    private var peakValues: [Float] = Array(repeating: 0, count: 75)
    private var peakVelocity: [Float] = Array(repeating: 0, count: 75)

    // MARK: - Rust handle

    // Rust ManzoHandle pointer — set by AppDelegate after manzo_open succeeds.
    // Optional<UnsafeMutablePointer<manzo_ManzoHandle>> — matches AppDelegate.manzoHandle type.
    var manzoHandle: UnsafeMutablePointer<manzo_ManzoHandle>? = nil

    // MARK: - Render loop

    private var displayLink: CADisplayLink?
    private var backgroundThread: Thread?
    private var backgroundRunLoop: RunLoop?

    // MARK: - Initialization

    init() {
        // Obtain the default Metal device.
        guard let device = MTLCreateSystemDefaultDevice() else {
            fatalError("MANZO Phase 7: MTLCreateSystemDefaultDevice() returned nil — Metal not available")
        }
        // Frame: 225 × 32 pt (D-01: 75 bars × 3 pt/bar, 16 height levels × 2 pt/level)
        super.init(frame: NSRect(x: 0, y: 0, width: 225, height: 32), device: device)
        self.device = device

        configureMetal()
        setupBuffers()
        setupShaders()

        NSLog("MANZO Phase 7: ManzoSpectrumView initialized — 225×32 pt, .rgba16Float, displayP3")
    }

    required init(coder: NSCoder) { fatalError("init(coder:) not used — code-only UI") }

    // MARK: - Metal Configuration

    private func configureMetal() {
        // CRITICAL (SPEC-01 / spike 010 landmine):
        // MTKView.colorPixelFormat does NOT configure the CAMetalLayer directly on macOS
        // when using a custom render loop — set CAMetalLayer fields explicitly.
        guard let metalLayer = self.layer as? CAMetalLayer else {
            fatalError("MANZO Phase 7: MTKView.layer is not a CAMetalLayer")
        }
        // .rgba16Float: supports values > 1.0 for HDR bloom highlights (spike 011).
        // Setting on CAMetalLayer directly ensures the drawable has the correct format
        // regardless of what MTKView's internal state tracks.
        metalLayer.pixelFormat = .rgba16Float
        // Explicit P3 colorspace — silent sRGB fallback if omitted (spike 010 landmine).
        metalLayer.colorspace = CGColorSpace(name: CGColorSpace.displayP3)
        // EDR: values slightly > 1.0 render as true HDR highlights on ProMotion displays.
        metalLayer.wantsExtendedDynamicRangeContent = true

        // MTKView configuration: disable automatic rendering — we drive the loop ourselves.
        self.isPaused = true
        self.enableSetNeedsDisplay = false

        // Transparent background so bodyView chrome shows through the edges.
        self.clearColor = MTLClearColorMake(0, 0, 0, 0)

        // Layer: transparent NSView backing (matches project constraint: no nested NSVisualEffectView).
        self.wantsLayer = true
        // Do NOT set shouldRasterize on the MTKView — it renders live (CONTEXT.md established patterns).
    }

    private func setupBuffers() {
        guard let device = self.device else { return }

        // fftBuffer: 75 Float values — storageModeShared for CPU+GPU access without copy (SPEC-02).
        fftBuffer = device.makeBuffer(length: 75 * MemoryLayout<Float>.stride, options: .storageModeShared)!
        // Zero-initialize: shows a flat spectrum until FFT data arrives.
        memset(fftBuffer.contents(), 0, 75 * MemoryLayout<Float>.stride)

        // barVertexBuffer: 75 bars × 6 vertices (2 triangles per bar quad).
        // Each vertex: float2 position (NDC) + float barIndex + float barHeight = 4 floats = 16 bytes.
        barVertexBuffer = device.makeBuffer(length: 75 * 6 * 4 * MemoryLayout<Float>.stride, options: .storageModeShared)!

        // peakVertexBuffer: 75 peak dots × 6 vertices × 4 floats.
        peakVertexBuffer = device.makeBuffer(length: 75 * 6 * 4 * MemoryLayout<Float>.stride, options: .storageModeShared)!
    }

    private func setupShaders() {
        guard let device = self.device else { return }

        // Load the default Metal library (Spectrum.metal is compiled into it by Xcode).
        guard let library = device.makeDefaultLibrary() else {
            fatalError("MANZO Phase 7: makeDefaultLibrary() returned nil — Spectrum.metal not compiled")
        }

        let barVertex   = library.makeFunction(name: "barVertex")!
        let barFragment = library.makeFunction(name: "barFragment")!
        let peakVertex   = library.makeFunction(name: "peakVertex")!
        let peakFragment = library.makeFunction(name: "peakFragment")!

        // Bar pipeline descriptor.
        let barDesc = MTLRenderPipelineDescriptor()
        barDesc.vertexFunction   = barVertex
        barDesc.fragmentFunction = barFragment
        barDesc.colorAttachments[0].pixelFormat = .rgba16Float
        // Additive blending for glow accumulation: src*srcAlpha + dst*1
        barDesc.colorAttachments[0].isBlendingEnabled = true
        barDesc.colorAttachments[0].sourceRGBBlendFactor = .sourceAlpha
        barDesc.colorAttachments[0].destinationRGBBlendFactor = .one
        barDesc.colorAttachments[0].sourceAlphaBlendFactor = .sourceAlpha
        barDesc.colorAttachments[0].destinationAlphaBlendFactor = .one
        barPipelineState = try! device.makeRenderPipelineState(descriptor: barDesc)

        // Peak pipeline descriptor (same blending, different shaders).
        let peakDesc = MTLRenderPipelineDescriptor()
        peakDesc.vertexFunction   = peakVertex
        peakDesc.fragmentFunction = peakFragment
        peakDesc.colorAttachments[0].pixelFormat = .rgba16Float
        peakDesc.colorAttachments[0].isBlendingEnabled = true
        peakDesc.colorAttachments[0].sourceRGBBlendFactor = .sourceAlpha
        peakDesc.colorAttachments[0].destinationRGBBlendFactor = .one
        peakDesc.colorAttachments[0].sourceAlphaBlendFactor = .sourceAlpha
        peakDesc.colorAttachments[0].destinationAlphaBlendFactor = .one
        peakPipelineState = try! device.makeRenderPipelineState(descriptor: peakDesc)
    }

    // MARK: - Render Loop Start/Stop (SPEC-04)

    // Call this from AppDelegate after manzo_open + manzo_play succeeds.
    func startRenderLoop() {
        guard displayLink == nil else { return }

        // Launch background thread with its own RunLoop.
        let thread = Thread {
            let runLoop = RunLoop.current
            self.backgroundRunLoop = runLoop

            // CADisplayLink on background thread — main thread is never blocked by GPU work (SPEC-04).
            let link = CADisplayLink(target: self, selector: #selector(self.displayLinkFired))
            link.add(to: runLoop, forMode: .common)
            self.displayLink = link

            NSLog("MANZO Phase 7: CADisplayLink render loop started on background thread")
            runLoop.run() // runs until stopped via invalidate + CFRunLoopStop
        }
        thread.name = "ManzoSpectrumRender"
        thread.qualityOfService = .userInteractive
        thread.start()
        backgroundThread = thread
    }

    func stopRenderLoop() {
        displayLink?.invalidate()
        displayLink = nil
        if let rl = backgroundRunLoop {
            CFRunLoopStop(rl.getCFRunLoop())
        }
        NSLog("MANZO Phase 7: CADisplayLink render loop stopped")
    }

    // MARK: - Per-Frame Render

    @objc private func displayLinkFired() {
        // 1. Read FFT data from Rust core into the shared MTLBuffer.
        //    manzo_get_spectrum writes 75 floats; range [0.0, 1.0] per bar.
        let fftPtr = fftBuffer.contents().assumingMemoryBound(to: Float.self)
        manzo_get_spectrum(manzoHandle, fftPtr, 75)

        // 2. Update peak-hold state (D-15: if fft[i] > peak, snap up; else decay).
        //    spfo = 1.1 (config_sa_peak_falloff=1, Winamp canonical).
        let spfo: Float = 1.1
        for i in 0..<75 {
            let fftVal = fftPtr[i]
            if fftVal > peakValues[i] {
                peakValues[i] = fftVal
                peakVelocity[i] = 3.0 / 256.0  // reset velocity on snap-up (D-15)
            } else {
                peakVelocity[i] *= spfo
                peakValues[i] = max(0.0, peakValues[i] - peakVelocity[i])
            }
        }

        // 3. Build bar vertex buffer (75 bars, 6 vertices each, extended ±6 pt for SDF bloom).
        buildBarVertices()
        buildPeakVertices()

        // 4. Encode and submit the render command.
        render()
    }

    // MARK: - Vertex Building

    // NDC helper: converts pt from [0, viewWidth/viewHeight] to [-1, 1].
    // MTKView origin is bottom-left in Metal NDC; AppKit is top-left.
    // We build in Metal NDC directly: x in [-1,1], y in [-1,1] (bottom-left origin).

    private func buildBarVertices() {
        let ptr = barVertexBuffer.contents().assumingMemoryBound(to: Float.self)
        let viewW: Float = 225.0
        let viewH: Float = 32.0
        let sigma: Float = 2.0          // SDF bloom sigma for bars
        let extend: Float = 3.0 * sigma  // 6.0 pt extension on all sides

        var vtxIdx = 0
        for bar in 0..<75 {
            let fftPtr = fftBuffer.contents().assumingMemoryBound(to: Float.self)
            let barH = fftPtr[bar]  // normalized height [0.0, 1.0]

            // Bar geometry in pt (bottom-left origin, y=0 at bottom of view).
            let barWidthPt: Float = 2.0
            let gapPt: Float = 1.0
            let xLeft   = Float(bar) * (barWidthPt + gapPt)
            let xRight  = xLeft + barWidthPt
            let yBottom: Float = 0.0
            let yTop    = barH * viewH  // bar top in pts

            // Extend all sides by 6 pt for the bloom halo.
            let xl = xLeft  - extend
            let xr = xRight + extend
            let yb = yBottom - extend
            let yt = yTop    + extend

            // Convert to NDC: x: [0,225]→[-1,1], y: [0,32]→[-1,1]
            func ndcX(_ x: Float) -> Float { (x / viewW) * 2.0 - 1.0 }
            func ndcY(_ y: Float) -> Float { (y / viewH) * 2.0 - 1.0 }

            // Two triangles (6 vertices) forming a quad.
            // Per vertex: [ndcX, ndcY, barIndex, barHeight] — 4 floats.
            let bi = Float(bar)
            let verts: [(Float, Float, Float, Float)] = [
                (ndcX(xl), ndcY(yb), bi, barH),  // bottom-left
                (ndcX(xr), ndcY(yb), bi, barH),  // bottom-right
                (ndcX(xl), ndcY(yt), bi, barH),  // top-left
                (ndcX(xr), ndcY(yb), bi, barH),  // bottom-right
                (ndcX(xr), ndcY(yt), bi, barH),  // top-right
                (ndcX(xl), ndcY(yt), bi, barH),  // top-left
            ]
            for v in verts {
                ptr[vtxIdx + 0] = v.0
                ptr[vtxIdx + 1] = v.1
                ptr[vtxIdx + 2] = v.2
                ptr[vtxIdx + 3] = v.3
                vtxIdx += 4
            }
        }
    }

    private func buildPeakVertices() {
        let ptr = peakVertexBuffer.contents().assumingMemoryBound(to: Float.self)
        let viewW: Float = 225.0
        let viewH: Float = 32.0
        let sigma: Float = 4.0    // peak dots: sigma × 2.0 = stronger glow (D-16)
        let extend: Float = 3.0 * sigma  // 12.0 pt extension

        var vtxIdx = 0
        for bar in 0..<75 {
            let peakH = peakValues[bar]

            let barWidthPt: Float = 2.0
            let gapPt: Float = 1.0
            let xLeft  = Float(bar) * (barWidthPt + gapPt)
            let xRight = xLeft + barWidthPt
            let yDot   = peakH * viewH  // 1-pt dot centered on peak level

            let xl = xLeft  - extend
            let xr = xRight + extend
            let yb = yDot   - extend - 0.5  // ±0.5 for 1-pt dot height
            let yt = yDot   + extend + 0.5

            func ndcX(_ x: Float) -> Float { (x / viewW) * 2.0 - 1.0 }
            func ndcY(_ y: Float) -> Float { (y / viewH) * 2.0 - 1.0 }

            let bi = Float(bar)
            let verts: [(Float, Float, Float, Float)] = [
                (ndcX(xl), ndcY(yb), bi, peakH),
                (ndcX(xr), ndcY(yb), bi, peakH),
                (ndcX(xl), ndcY(yt), bi, peakH),
                (ndcX(xr), ndcY(yb), bi, peakH),
                (ndcX(xr), ndcY(yt), bi, peakH),
                (ndcX(xl), ndcY(yt), bi, peakH),
            ]
            for v in verts {
                ptr[vtxIdx + 0] = v.0
                ptr[vtxIdx + 1] = v.1
                ptr[vtxIdx + 2] = v.2
                ptr[vtxIdx + 3] = v.3
                vtxIdx += 4
            }
        }
    }

    // MARK: - Render

    private func render() {
        guard let drawable = currentDrawable,
              let commandQueue = commandQueue,
              let descriptor = currentRenderPassDescriptor else { return }

        descriptor.colorAttachments[0].loadAction = .clear
        descriptor.colorAttachments[0].clearColor = MTLClearColorMake(0, 0, 0, 0)

        guard let cmdBuf = commandQueue.makeCommandBuffer(),
              let encoder = cmdBuf.makeRenderCommandEncoder(descriptor: descriptor) else { return }

        // Pass 1: Draw spectrum bars with SDF bloom.
        encoder.setRenderPipelineState(barPipelineState)
        encoder.setVertexBuffer(barVertexBuffer, offset: 0, index: 0)
        encoder.drawPrimitives(type: .triangle, vertexStart: 0, vertexCount: 75 * 6)

        // Pass 2: Draw peak dots with stronger bloom (D-16).
        encoder.setRenderPipelineState(peakPipelineState)
        encoder.setVertexBuffer(peakVertexBuffer, offset: 0, index: 0)
        encoder.drawPrimitives(type: .triangle, vertexStart: 0, vertexCount: 75 * 6)

        encoder.endEncoding()
        cmdBuf.present(drawable)
        cmdBuf.commit()
    }

    // MARK: - MTKView commandQueue wiring

    override func viewDidMoveToWindow() {
        super.viewDidMoveToWindow()
        // Create commandQueue lazily when device is confirmed available.
        if commandQueue == nil, let device = self.device {
            commandQueue = device.makeCommandQueue()
        }
    }
}

#include <metal_stdlib>
using namespace metal;

// ─────────────────────────────────────────────────────────────────────────────
// Shared vertex input structure.
// Each vertex: [NDC position (float2), barIndex (float), barHeight (float)]
// barHeight is in [0.0, 1.0] — normalized height for this bar (D-05).
// ─────────────────────────────────────────────────────────────────────────────
struct BarVertex {
    float2 position [[attribute(0)]];
    float  barIndex  [[attribute(1)]];
    float  barHeight [[attribute(2)]];
};

struct BarVaryings {
    float4 position [[position]];
    float2 uv;         // fragment position in pt-space (used for SDF distance)
    float  barIndex;
    float  barHeight;
    float  sigma;      // glow sigma — passed from vertex to fragment
};

// ─────────────────────────────────────────────────────────────────────────────
// specColor: 5-stop ppal2 gradient with P3 neon upgrades (D-12).
// h = normalized bar height [0.0=bottom, 1.0=top].
// These are the canonical values — DO NOT round or approximate.
// ─────────────────────────────────────────────────────────────────────────────
float4 specColor(float h) {
    float4 c0 = float4(0.094, 0.518, 0.031, 1.0); // ppal2[17] dark green (bottom)
    float4 c1 = float4(0.000, 0.950, 0.060, 1.0); // ppal2[12] → P3 neon green
    float4 c2 = float4(0.741, 0.871, 0.161, 1.0); // ppal2[10] yellow-green
    float4 c3 = float4(0.920, 0.750, 0.000, 1.0); // ppal2[8–9] → P3 vivid gold
    float4 c4 = float4(1.000, 0.120, 0.040, 1.0); // ppal2[2] → P3 saturated red (top)
    if      (h < 0.35) return mix(c0, c1, h / 0.35);
    else if (h < 0.55) return mix(c1, c2, (h - 0.35) / 0.20);
    else if (h < 0.70) return mix(c2, c3, (h - 0.55) / 0.15);
    else               return mix(c3, c4, (h - 0.70) / 0.30);
}

// ─────────────────────────────────────────────────────────────────────────────
// Bar vertex shader.
// Input buffer layout (per vertex, 4 floats = 16 bytes):
//   [0] float ndcX
//   [1] float ndcY
//   [2] float barIndex
//   [3] float barHeight
// ─────────────────────────────────────────────────────────────────────────────
vertex BarVaryings barVertex(
    uint vid [[vertex_id]],
    const device float4* verts [[buffer(0)]])
{
    float4 v = verts[vid];

    BarVaryings out;
    out.position  = float4(v.x, v.y, 0.0, 1.0);
    // uv: encode position in a space where we can compute SDF distance.
    // We pass NDC directly; fragment computes screen-space distance via the SDF.
    out.uv        = float2(v.x, v.y);
    out.barIndex  = v.z;
    out.barHeight = v.w;
    out.sigma     = 2.0;  // bar bloom sigma (spike 011)
    return out;
}

// ─────────────────────────────────────────────────────────────────────────────
// Bar fragment shader — SDF single-pass bloom (SPEC-03, spike 011).
//
// The bar quad was extended by 3σ=6 pt on all sides in vertex building.
// The fragment receives the pixel's position in NDC space. To compute the
// distance from the bar edge, we use the barHeight to reconstruct the bar's
// top edge in NDC, and the constant bar bottom = -1.0 (NDC bottom of view).
//
// NDC mapping: view is 225 × 32 pt. Bar top in pt = barHeight * 32.
// In NDC: top_y = (barHeight * 32 / 32) * 2 - 1 = barHeight * 2 - 1.
//
// Distance d = distance from fragment to the nearest bar edge (4 edges).
// SDF glow: exp(-d*d / (2*sigma*sigma))
// Output: barColor * glow   (halo)  + barColor * step(0, 0.5-d) (solid core inside 0.5pt)
// ─────────────────────────────────────────────────────────────────────────────
fragment float4 barFragment(BarVaryings in [[stage_in]]) {
    float sigma   = in.sigma;
    float barH    = in.barHeight;

    // Bar edges in NDC space (view: 225×32 pt, NDC: [-1,1]×[-1,1]).
    float barWidthNDC = (2.0 / 225.0) * 2.0;  // 2 pt bar in NDC
    float gapNDC      = (1.0 / 225.0) * 2.0;  // 1 pt gap in NDC
    float xLeft  = in.barIndex * (barWidthNDC + gapNDC) - 1.0;
    float xRight = xLeft + barWidthNDC;
    float yBottom = -1.0;  // NDC bottom
    float yTop    = barH * 2.0 - 1.0;  // NDC top of bar

    // Fragment position in NDC.
    float px = in.uv.x;
    float py = in.uv.y;

    // Signed distance from rectangle [xLeft, xRight] × [yBottom, yTop].
    // Positive outside the rect, negative inside (standard SDF rect formula).
    float2 d2 = float2(
        max(xLeft - px, px - xRight),
        max(yBottom - py, py - yTop)
    );
    float d = length(max(d2, 0.0)) + min(max(d2.x, d2.y), 0.0);

    // Skip fragments far beyond the glow radius (> 4σ from edge) — transparent.
    if (d > 4.0 * sigma * (2.0 / 225.0)) discard_fragment();

    // Convert d from NDC to approximate pt distance.
    // 2.0 NDC units = 225 pt, so 1 NDC unit = 112.5 pt.
    float d_pt = d * 112.5;

    // Analytical Gaussian glow (spike 011 validated pattern).
    float glow = exp(-d_pt * d_pt / (2.0 * sigma * sigma));

    // Color: use height-based gradient, sampled at fragment's y position within bar.
    // Clamp to [0,1]: outside the bar (d>0) we still color the halo correctly.
    float fragH = saturate((py - yBottom) / max(yTop - yBottom, 0.001));
    float4 barColor = specColor(fragH);

    // For silent bars (barH == 0), skip entirely.
    if (barH < 0.01) return float4(0.0);

    // Output: halo glow + solid core (spike 011 formula).
    float core = step(d_pt, 0.5);  // solid inside 0.5 pt of edge
    return barColor * glow + barColor * core;
}

// ─────────────────────────────────────────────────────────────────────────────
// Peak dot vertex shader — same layout as bar vertex, different sigma.
// ─────────────────────────────────────────────────────────────────────────────
vertex BarVaryings peakVertex(
    uint vid [[vertex_id]],
    const device float4* verts [[buffer(0)]])
{
    float4 v = verts[vid];
    BarVaryings out;
    out.position  = float4(v.x, v.y, 0.0, 1.0);
    out.uv        = float2(v.x, v.y);
    out.barIndex  = v.z;
    out.barHeight = v.w;
    out.sigma     = 4.0;  // peak glow: σ × 2.0 = 4.0 (D-16)
    return out;
}

// ─────────────────────────────────────────────────────────────────────────────
// Peak dot fragment shader.
// Color: ppal2[23] RGB(150,150,150) → float4(0.588, 0.588, 0.588, 1.0) (D-13).
// Stronger glow (σ=4.0) per D-16. Same SDF formula as bars.
// ─────────────────────────────────────────────────────────────────────────────
fragment float4 peakFragment(BarVaryings in [[stage_in]]) {
    float sigma  = in.sigma;  // 4.0 — peak sigma
    float peakH  = in.barHeight;

    // Peak dot: 1-pt tall, 2-pt wide, at peakH height.
    float barWidthNDC = (2.0 / 225.0) * 2.0;
    float gapNDC      = (1.0 / 225.0) * 2.0;
    float xLeft  = in.barIndex * (barWidthNDC + gapNDC) - 1.0;
    float xRight = xLeft + barWidthNDC;
    float dotY   = peakH * 2.0 - 1.0;  // NDC center of peak dot
    float dotHalfH = (0.5 / 32.0) * 2.0;  // 0.5 pt in NDC
    float yBottom = dotY - dotHalfH;
    float yTop    = dotY + dotHalfH;

    float px = in.uv.x;
    float py = in.uv.y;

    float2 d2 = float2(
        max(xLeft - px, px - xRight),
        max(yBottom - py, py - yTop)
    );
    float d    = length(max(d2, 0.0)) + min(max(d2.x, d2.y), 0.0);
    float d_pt = d * 112.5;

    // Skip far fragments.
    if (d_pt > 4.0 * sigma) discard_fragment();
    if (peakH < 0.01) return float4(0.0);

    float glow = exp(-d_pt * d_pt / (2.0 * sigma * sigma));
    float core = step(d_pt, 0.5);

    // D-13: peak dot color = ppal2[23] RGB(150,150,150) — neutral gray, no P3 extension.
    float4 dotColor = float4(0.588, 0.588, 0.588, 1.0);
    return dotColor * glow + dotColor * core;
}
